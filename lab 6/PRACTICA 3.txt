================================================================================
GUÍA DEFINITIVA: INSTALACIÓN, CONFIGURACIÓN Y EJECUCIÓN DE SNORT 3 EN RHEL 10
================================================================================

1. RESOLUCIÓN DE REPOS DEPURACIÓN Y EPEL EN RHEL 10
--------------------------------------------------------------------------------
# Remover EPEL 9 si estaba presente por error y configurar EPEL 10 oficial
sudo dnf remove -y epel-release
sudo dnf install -y https://dl.fedoraproject.org/pub/epel/epel-release-latest-10.noarch.rpm --nogpgcheck

# Habilitar el repositorio exacto de CodeReady Linux Builder (CRB) en RHEL 10
sudo subscription-manager repos --enable=codeready-builder-for-rhel-10-x86_64-rpms
sudo dnf clean all

2. INSTALACIÓN DE DEPENDENCIAS COMPLETAS DEL SISTEMA
--------------------------------------------------------------------------------
# Herramientas de desarrollo base y compiladores
sudo dnf -y groupinstall "Development Tools" --nogpgcheck
sudo dnf -y install cmake gcc gcc-c++ make git --nogpgcheck

# Librerías necesarias para Snort 3, libdaq y decodificación de paquetes
sudo dnf -y install libpcap-devel pcre2-devel openssl-devel zlib-devel zlib-ng-compat-devel luajit-devel libdnet libdnet-devel hwloc hwloc-devel libnghttp2-devel --nogpgcheck

--------------------------------------------------------------------------------
* NOTA EN CASO DE QUE 'libdnet-devel' NO SEA ENCONTRADO POR DNF:
  Compilar e instalar libdnet manualmente desde el repositorio oficial:
  cd /usr/src
  sudo git clone https://github.com/ofalk/libdnet.git
  cd libdnet
  sudo ./configure && sudo make -j$(nproc) && sudo make install
  sudo ln -sf /usr/local/include/dnet.h /usr/include/dnet.h
  sudo ldconfig
--------------------------------------------------------------------------------

3. COMPILACIÓN E INSTALACIÓN DE LIBDAQ (DATA ACQUISITION LIBRARY)
--------------------------------------------------------------------------------
cd /usr/src
sudo git clone https://github.com/snort3/libdaq.git
cd libdaq
sudo ./bootstrap
sudo ./configure
sudo make -j$(nproc)
sudo make install

4. CONFIGURACIÓN DEL ENTORNO Y COMPILACIÓN DE SNORT 3
--------------------------------------------------------------------------------
export PKG_CONFIG_PATH=/usr/local/lib/pkgconfig:$PKG_CONFIG_PATH

cd /usr/src
sudo git clone https://github.com/snort3/snort3.git
cd snort3
sudo rm -rf build/
sudo -E ./configure_cmake.sh --prefix=/usr/local
cd build
sudo make -j$(nproc)
sudo make install

5. CONFIGURACIÓN GLOBAL DE LIBRERÍAS (SOLUCIÓN DEFINITIVA ERRORES DE ENLAZADO)
--------------------------------------------------------------------------------
echo "/usr/local/lib" | sudo tee /etc/ld.so.conf.d/snort.conf
sudo ldconfig

6. CREACIÓN DE REGLAS DE DETECCIÓN PERSONALIZADAS (LOCAL.RULES)
--------------------------------------------------------------------------------
sudo mkdir -p /usr/local/etc/snort/rules

sudo tee /usr/local/etc/snort/rules/local.rules << 'EOF'
# 1. Detectar todo el trafico ICMP (Ping) hacia el host
alert icmp any any -> any any (msg:"ALERTA: Trafico ICMP (Ping) Detectado"; sid:1000001; rev:1;)

# 2. Detectar todo el trafico hacia el puerto 21/tcp (FTP)
alert tcp any any -> any 21 (msg:"ALERTA: Intento de conexion FTP (Puerto 21)"; sid:1000002; rev:1;)

# 3. Detectar todo el trafico hacia el puerto 22/tcp (SSH)
alert tcp any any -> any 22 (msg:"ALERTA: Intento de conexion SSH (Puerto 22)"; sid:1000003; rev:1;)

# 4. Detectar todo el trafico hacia el puerto 80/tcp (HTTP)
alert tcp any any -> any 80 (msg:"ALERTA: Intento de conexion HTTP (Puerto 80)"; sid:1000004; rev:1;)
EOF

7. VINCULACIÓN INYECCIÓN LIMPIA EN SNORT.LUA (EVITA ERRORES DE SINTAXIS LUA)
--------------------------------------------------------------------------------
sudo tee -a /usr/local/etc/snort/snort.lua << 'EOF'

ips = {
    variables = default_variables,
    include = '/usr/local/etc/snort/rules/local.rules'
}
EOF

8. VALIDACIÓN DE LA CONFIGURACIÓN (PRUEBA DE SINTAXIS)
--------------------------------------------------------------------------------
sudo /usr/local/bin/snort -c /usr/local/etc/snort/snort.lua -T

9. EJECUCIÓN EN MODO ESCUCHA (REEMPLAZAR ens160 POR TU INTERFAZ SI ES DISTINTA)
--------------------------------------------------------------------------------
sudo /usr/local/bin/snort -c /usr/local/etc/snort/snort.lua -i ens160 -A alert_fast
